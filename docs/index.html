🦜 LangChain 1.x Tool Calling Python 
from langchain_openai import ChatOpenAI
from langchain_opticparse import OpticParseTool, PhishVisionTool

# 1. Initialize Tools (inherits from BaseTool)
scraper = OpticParseTool()
scanner = PhishVisionTool()

# 2. Bind directly to any Tool-Calling LLM
llm = ChatOpenAI(model="gpt-4o", temperature=0)
llm_with_tools = llm.bind_tools([scraper, scanner])

# 3. Direct Tool Invocation Example
data = scraper.invoke({
    "url": "https://news.ycombinator.com",
    "query": "Extract top 5 stories with points and author"
})
print("Extracted Data:", data)
